const StepCard: React.FC<{ step: OnboardingStep; index: number }> = ({ step, index }) => {
  const isNextAction = !step.completed && !step.locked;
  
  return (
    <div style={{
      backgroundColor: '#ffffff',
      border: `1px solid ${isNextAction ? '#171717' : '#ebebeb'}`,
      borderRadius: '12px',

      display: 'flex',
      alignItems: 'flex-start',
      gap: '16px',
      opacity: step.locked ? 0.5 : 1,
    }}>
      {/* Status indicator */}
      <div style={{
        width: '24px',
        height: '24px',
        borderRadius: '9999px',
        border: `2px solid ${step.completed ? '#171717' : step.locked ? '#ebebeb' : '#171717'}`,
        backgroundColor: step.completed ? '#171717' : '#ffffff',
        display: 'flex',
        alignItems: 'center',
        justifyContent: 'center',
        flexShrink: 0,

      </div>

      {/* Content */}
      <div style={{ flex: 1, minWidth: 0 }}>
        <h3 style={{
          fontSize: '16px',
          lineHeight: '24px',
          fontWeight: 600,
          color: step.locked ? '#8f8f8f' : '#171717',

            margin: '0 0 16px 0',
          }}>{step.description}</p>
        )}

        {/* Actions */}

          <div style={{
            display: 'flex',

            gap: '12px',

          }}>
